\makeproblem{APLUSB}{Bài toán A + B}{1.0s}{256 MB}

Cho hai số nguyên $a$ và $b$. Hãy tính tổng của chúng.

\medskip
\textbf{Yêu cầu:} In ra giá trị của $a + b$.

\makesection{Input}
\begin{itemize}
    \item Gồm một dòng duy nhất chứa hai số nguyên $a$ và $b$, cách nhau bởi một dấu cách.
\end{itemize}

\makesection{Output}
\begin{itemize}
    \item In ra một số nguyên duy nhất là kết quả của phép tính $a + b$.
\end{itemize}

\begin{makesubtasks}
    \subtask{40} $|a|, |b| \le 100$.
    \subtask{60} $|a|, |b| \le 10^9$.
\end{makesubtasks}

\makesection{Ví dụ}

\subsubsection*{Ví dụ 1}
\begin{makesample}
\begin{lstlisting}
5 7
\end{lstlisting}
\nextsample
\begin{lstlisting}
12
\end{lstlisting}
\end{makesample}

\begin{explain}
    Tổng của hai số $5$ và $7$ bằng $12$.
\end{explain}

\subsubsection*{Ví dụ 2}
\begin{makesample}
\begin{lstlisting}
-3 8
\end{lstlisting}
\nextsample
\begin{lstlisting}
5
\end{lstlisting}
\end{makesample}

\begin{explain}
    Tổng của hai số $-3$ và $8$ bằng $5$.
\end{explain}
